cmake_minimum_required(VERSION 3.16)
project(shar_shader_interface_probe LANGUAGES C CXX)
set(MGL_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../src/mgl" CACHE PATH "Restored MGL source checkout")
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_C_API OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_CPP OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_REFLECT OFF CACHE BOOL "" FORCE)
add_subdirectory("${MGL_SOURCE}/external/glslang" glslang)
add_subdirectory("${MGL_SOURCE}/external/SPIRV-Cross" cross)
add_executable(shader_interface_probe shader_interface_probe.cpp)
target_compile_features(shader_interface_probe PRIVATE cxx_std_17)
target_include_directories(shader_interface_probe PRIVATE "${MGL_SOURCE}/external/glslang")
target_link_libraries(shader_interface_probe PRIVATE glslang glslang-default-resource-limits spirv-cross-msl)
